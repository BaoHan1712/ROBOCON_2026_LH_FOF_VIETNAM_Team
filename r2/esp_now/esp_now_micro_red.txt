#include <WiFi.h>
#include <esp_now.h>

#define DEVICE_NAME "ESP32-B"

// ================== Cấu hình UART cho Nextion ==================
#define NEXTION_TX 17   // TXD1 → RX của Nextion
#define NEXTION_RX 18   // RXD1 → TX của Nextion
#define NEXTION_BAUD 9600

HardwareSerial nextionSerial(1);  // dùng UART1

// ============= Hàm gửi chuỗi đến Nextion (.txt="...") ============
void sendToNextion(const char *txt) {
  // Ví dụ: hiển thị lên text box t0 trên HMI
  nextionSerial.print("t0.txt=\"");
  nextionSerial.print(txt);
  nextionSerial.print("\"");
  // 3 ký tự kết thúc lệnh Nextion
  nextionSerial.write(0xFF);
  nextionSerial.write(0xFF);
  nextionSerial.write(0xFF);
}

// ================= Callback khi nhận dữ liệu ESP-NOW =================
void onDataRecv(const esp_now_recv_info_t *info, const uint8_t *incomingData, int len) {
  char recvMsg[200];
  memcpy(recvMsg, incomingData, len);
  recvMsg[len] = '\0';  // đảm bảo chuỗi kết thúc

  const uint8_t *mac = info->src_addr;
  Serial.printf("📡 From %02X:%02X:%02X:%02X:%02X:%02X → %s\n",
                mac[0], mac[1], mac[2],
                mac[3], mac[4], mac[5],
                recvMsg);

  // Gửi nội dung này lên màn hình Nextion
  sendToNextion(recvMsg);
}

// ================= Setup =================
void setup() {
  Serial.begin(115200);                      // debug USB
  nextionSerial.begin(NEXTION_BAUD, SERIAL_8N1, NEXTION_RX, NEXTION_TX);  // UART1 cho Nextion

  WiFi.mode(WIFI_STA);

  Serial.printf("[%s] MAC: %s\n", DEVICE_NAME, WiFi.macAddress().c_str());

  if (esp_now_init() != ESP_OK) {
    Serial.println("❌ ESP-NOW init failed!");
    return;
  }

  esp_now_register_recv_cb(onDataRecv);
  Serial.println("✅ ESP32-B sẵn sàng nhận dữ liệu & gửi Nextion!");
}

// ================= Loop =================
void loop() {
  // Không cần xử lý thêm
}
